CREATE TABLE address_5a (
    id INT PRIMARY KEY AUTO_INCREMENT,
    streetAddress VARCHAR(255),
    City VARCHAR(100),
    State VARCHAR(100),
    ZipCode VARCHAR(20)
);

CREATE TABLE user_5a (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    email VARCHAR(150),
    homeAddressId INT,
    officeAddressId INT,

    FOREIGN KEY (homeAddressId) REFERENCES address_5a(id),
    FOREIGN KEY (officeAddressId) REFERENCES address_5a(id)
);
